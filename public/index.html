function startTriviaGame() { score = 0; round = 1; currentQIndex = 0; document.getElementById('score-indicator').innerText = 'Score: 0'; goToScreen('game-screen'); showQuestion(); } function showQuestion() { if (currentQIndex >= sampleQuestions.length) { endGame(); return; } const question = sampleQuestions[currentQIndex]; document.getElementById('round-indicator').innerText = `ROUND ${currentQIndex + 1} OF ${sampleQuestions.length}`; document.getElementById('game-category').innerText = question.cat; document.getElementById('question-text').innerText = question.q; const container = document.getElementById('options-container'); container.innerHTML = ''; document.getElementById('next-btn').style.display = 'none'; question.o.forEach((option, index) => { const button = document.createElement('button'); button.className = 'option-btn'; button.innerText = option; button.onclick = function () { checkAnswer(index, button); }; container.appendChild(button); }); startTimer(); } let timerInterval; let timeLeft = 15; function startTimer() { clearInterval(timerInterval); timeLeft = 15; const timerFill = document.getElementById('timer-bar-fill'); timerFill.style.width = '100%'; timerInterval = setInterval(() => { timeLeft--; timerFill.style.width = `${(timeLeft / 15) * 100}%`; if (timeLeft <= 0) { clearInterval(timerInterval); revealAnswer(); } }, 1000); } function checkAnswer(selectedIndex, button) { clearInterval(timerInterval); const question = sampleQuestions[currentQIndex]; const buttons = document.querySelectorAll('.option-btn'); buttons.forEach(btn => { btn.disabled = true; btn.style.cursor = 'default'; }); if (selectedIndex === question.a) { score += 100 + Math.max(0, timeLeft * 5); button.classList.add('correct'); button.style.borderColor = '#10b981'; } else { button.style.background = 'rgba(239, 68, 68, 0.2)'; button.style.borderColor = '#ef4444'; } buttons[question.a].classList.add('correct'); document.getElementById('score-indicator').innerText = `Score: ${score}`; document.getElementById('next-btn').style.display = 'block'; } function revealAnswer() { const question = sampleQuestions[currentQIndex]; const buttons = document.querySelectorAll('.option-btn'); buttons.forEach(btn => { btn.disabled = true; btn.style.cursor = 'default'; }); if (buttons[question.a]) { buttons[question.a].classList.add('correct'); } document.getElementById('next-btn').style.display = 'block'; } function nextQuestion() { clearInterval(timerInterval); currentQIndex++; round = currentQIndex + 1; showQuestion(); } function endGame() { clearInterval(timerInterval); goToScreen('end-screen'); document.getElementById('end-status').innerText = `You scored ${score} points!`; const leaderboard = document.getElementById('final-leaderboard'); leaderboard.innerHTML = ''; const nickname = document.getElementById('nickname-input').value.trim() || 'Player 1'; const players = [ { name: nickname, score: score }, ...demoPlayers ]; players.sort((a, b) => b.score - a.score); players.forEach((player, index) => { const li = document.createElement('li'); li.className = 'player-item'; const name = document.createElement('span'); name.innerText = `${index + 1}. ${player.name}`; const points = document.createElement('span'); points.innerText = `${player.score} pts`; li.appendChild(name); li.appendChild(points); leaderboard.appendChild(li); }); } function restartGame() { clearInterval(timerInterval); score = 0; round = 1; currentQIndex = 0; goToScreen('home-screen'); }
